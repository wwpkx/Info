# 安装 QEMU​
```
sudo apt install -y qemu qemu-kvm libvirt-daemon-system virt-manager qemu-utils qemu-system qemu-user-static qemu-user
• qemu：基础模拟器
• qemu-kvm：硬件加速模块（需 CPU 支持 VT-x/AMD-V）
• virt-manager：图形化管理工具（可选）

sudo apt install -y qemu qemu-system qemu-user-static qemu-utils
```

# 安装多架构支持
```
# ARM 架构支持
sudo apt install qemu-system-arm
# RISC-V 架构支持
sudo apt install qemu-system-riscv64
```

# 检查 QEMU 版本
```
qemu-system-x86_64 --version  # 验证 x86 模拟器
qemu-system-arm --version     # 验证 ARM 模拟器
qemu-arm --version         # 验证用户模式模拟器（用于直接运行 ARM 程序）
```

# 用户模式模拟
```
qemu-arm -L /usr/arm-linux-gnueabihf ./program_arm32
qemu-arm -L /usr/arm-linux-gnueabihf ./program_arm32_2
```

# 用 qemu-system-arm 启动树莓派镜像（指定硬件型号为树莓派3）
qemu-system-arm \

  -kernel kernel8.img \  # 树莓派3 需要单独的内核文件（见下方说明）

  -sd 2023-10-10-raspios-bookworm-armhf-lite.img \  # 系统镜像
  -append "console=ttyAMA0 root=/dev/mmcblk0p2 rw" \  # 启动参数
  -serial stdio \  # 串口输出到终端
  -display none  # 不启动图形界面（纯命令行）
