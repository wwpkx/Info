# 查看架构
```
uname -m  # x86_64
arch
/proc/cpuinfo
```
# 查看内核版本
```
uname -r
uname -a
cat /proc/version
```
# 编译 x86_64 程序
```bash
# 安装默认 GCC
sudo apt install -y gcc g++

# 编译 64位 C 程序
gcc -o program_x86_64 program.c

# 验证架构（应显示 x86-64）
# program_x86_64: ELF 64-bit LSB pie executable, x86-64 
file program_x86_64
```

# 编译 x86（32 位）程序
```bash
# 安装 32位编译依赖
sudo apt install -y gcc-multilib g++-multilib

# 编译 32位 C 程序（-m32 指定32位架构）
gcc -m32 -o program_x86 program.c

# 验证架构（应显示 Intel 80386 或 i386）
# program_x86: ELF 32-bit LSB pie executable, Intel 80386,
file program_x86
```

# 安装交叉编译工具链
```bash
# 32位 ARM（armhf，硬浮点）
# 支持硬浮点的 PIE 程序（适合现代 ARM 设备，性能高、安全性强）
sudo apt install -y gcc-arm-linux-gnueabihf g++-arm-linux-gnueabihf
# 编译 32位 ARM 程序（树莓派等设备适用）
arm-linux-gnueabihf-gcc -o program_arm32 program.c
# 验证架构（应显示 ARM）
# program_arm32: ELF 32-bit LSB pie executable, ARM
file program_arm32

# 32位 ARM（arm，软浮点）
# 支持软浮点的传统可执行程序（适合老旧设备，兼容性好但性能和安全性较弱）
sudo apt install -y gcc-arm-linux-gnueabi g++-arm-linux-gnueabi
# 编译 32位 ARM 程序（树莓派等设备适用）
arm-linux-gnueabi-gcc -o program_arm32_2 program.c
# 验证架构（应显示 ARM）
# program_arm32_2: ELF 32-bit LSB executable, ARM
file program_arm32_2

# 64位 ARM（aarch64）
sudo apt install -y gcc-aarch64-linux-gnu g++-aarch64-linux-gnu
# 编译 64位 ARM 程序（ARMv8 及以上设备）
aarch64-linux-gnu-gcc -o program_arm64 program.c
# 验证架构（应显示 ARM aarch64）
# program_arm64: ELF 64-bit LSB pie executable, ARM aarch64
file program_arm64
```
