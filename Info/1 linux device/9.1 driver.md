
# driver
- device_driver
	- 是一个**通用的驱动结构体**，定义在 include/linux/device/driver.h 中
	- 适用于所有类型的设备驱动（如 PCI、USB、Platform 等）。
- platform_driver
	- **是 device_driver 的一种特化**，专门用于 Platform 设备（嵌入式系统中常见的设备）。
	- 封装了 device_driver，并添加了与 Platform 设备相关的功能。
	- 它在 include/linux/platform_device.h 中定义

# device_driver
static struct device_driver s3c2410ts_driver = {
	.name    = "s3c2410-ts",			// 驱动程序的名称，用于标识该驱动程序
	.bus     = &platform_bus_type,		// 指定驱动程序所属的总线类型
	.probe   = s3c2410ts_probe,			// 当设备与驱动程序匹配时，内核会调用该函数来初始化设备
	.remove  = s3c2410ts_remove,		// 当设备被卸载或驱动程序被移除时，内核会调用该函数来清理资源
};

driver_register				// 用于平台设备驱动或总线驱动的注册
input_register_device	// 将设备注册到内核的**输入子系统**，使其能够被用户空间访问
input_report_abs		// 报告绝对轴事件（如触摸屏的 X/Y 坐标）
input_report_key		// 报告按键事件（如触摸屏的按下/松开）
input_sync				// 同步输入事件，通知输入子系统事件已完成
clk_disable			// 禁用时钟信号
clk_unuse			// 释放时钟资源的使用权
clk_get_rate		// 用于获取指定时钟源的频率
clk_get				// 用于获取一个时钟资源
clk_put				// 释放时钟资源

# platform_driver
static struct platform_driver s3c2410_rtcdrv = {
	.probe		= s3c2410_rtc_probe,    // 初始化驱动
	.remove		= s3c2410_rtc_remove,   // 移除或驱动卸载
	.shutdown   = 						// 系统关机或重启时调用，确保设备进入安全状态，例如关闭硬件、停止操作等
	.suspend	= s3c2410_rtc_suspend,  // 当系统进入低功耗模式（如挂起）时，内核会调用 suspend 函数
	.resume		= s3c2410_rtc_resume,   // 当系统从低功耗模式恢复时，内核会调用 resume 函数
	.driver		= {
		.name	= "s3c2410-rtc",        // 驱动的名称，必须与设备的名称匹配
		.owner	= THIS_MODULE,
	},
};

platform_driver_register	// 用于注册一个平台驱动（platform_driver），将驱动与设备匹配。
platform_get_irq			// 获取设备的中断号
platform_get_resource		// 获取设备的资源（如内存、IO端口等）

# rtc
static struct rtc_ops s3c2410_rtcops = {
	.owner		= THIS_MODULE,          // 表示该操作属于当前模块。
	.open		= s3c2410_rtc_open,     // 打开 RTC 设备时调用的函数。
	.release	= s3c2410_rtc_release,  // 释放 RTC 设备时调用的函数。
	.ioctl		= s3c2410_rtc_ioctl,    // 处理用户空间的 IO 控制命令。
	.read_time	= s3c2410_rtc_gettime,  // 读取当前时间的函数。
	.set_time	= s3c2410_rtc_settime,  // 设置当前时间的函数。
	.read_alarm	= s3c2410_rtc_getalarm, // 读取闹钟时间的函数。
	.set_alarm	= s3c2410_rtc_setalarm, // 设置闹钟时间的函数。
	.proc	    = s3c2410_rtc_proc,     // 用于生成 RTC 的 `/proc` 文件信息。
};

register_rtc		// 注册 RTC 驱动
unregister_rtc		// 注销 RTC 驱动
