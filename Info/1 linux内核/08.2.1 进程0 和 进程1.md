| 特性   | 进程 0（idle 进程）         | 进程 1（init 进程）          |
|------|-----------------------|------------------------|
| PID  | 0                     | 1                      |
| 运行态  | 内核态（Ring0）            | 用户态（Ring3）             |
| 创建方式 | 内核静态初始化（无fork）        | 进程 0 通过kernel_thread创建 |
| 栈结构  | 仅内核栈（无用户栈）            | 内核栈 + 用户栈（完整双栈结构）      |
| 核心角色 | CPU 空闲时的 idle 任务、调度兜底 | 系统初始化、用户态进程始祖          |
| 退出影响 | 不可退出（内核依赖其调度兜底）       | 退出触发kernel panic（系统崩溃） |
| 优先级  | 最低（nice=20）           | 普通优先级（可配置）             |

void rest_init(void) {
    int pid;
    // 1. 初始化进程0的task_struct（进程描述符）
    // 内核静态分配进程0的task_struct，设置PID=0，状态为TASK_RUNNING
    current->pid = 0;
    rcu_scheduler_starting(); // 启动RCU调度机制（进程同步核心）
    
    // 2. 创建进程1（init进程）：通过kernel_thread（内核线程创建函数）
    pid = kernel_thread(init, NULL, CLONE_FS | CLONE_SIGHAND);
    // 3. 创建kthreadd进程（内核线程管理进程）：负责后续内核线程的创建/销毁
    pid = kernel_thread(kthreadd, NULL, CLONE_FS | CLONE_FILES);
    
    // 4. 进程0进入idle循环（永久运行）
    cpu_startup_entry(CPUHP_ONLINE);
}

void cpu_idle_loop(void) {
    while (1) {
        // 1. 关闭本地中断，检查是否有紧急任务
        local_irq_disable();
        if (need_resched()) {
            local_irq_enable();
            schedule(); // 有其他进程可运行，触发调度
            continue;
        }
        local_irq_enable();
        
        // 2. CPU空闲，执行低功耗等待（如x86的hlt指令）
        arch_cpu_idle(); // 暂停CPU，直到有中断触发（如时钟中断）
    }
}
